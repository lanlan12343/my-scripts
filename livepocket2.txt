// ==UserScript==
// @name         LivePocket 姓名切换
// @version      2.0
// @description  在 LivePocket 上自定义显示姓名
// @author       andy lin
// @match        https://livepocket.jp/*
// @match        https://t.livepocket.jp/*
// @grant        GM_getValue
// @grant        GM_setValue
// @grant        GM_registerMenuCommand
// @run-at       document-start
// @noframes
// ==/UserScript==

(function() {
  'use strict';
  
  // ==================== Constants ====================
  const DATA_KEY = 'lp_display_data';
  const TOGGLE_KEY = 'lp_display_on';
  const FIELD_LIST = ['氏名', '生年月日', '性別'];
  const HEADER_SELECTOR = '.header-account__name span[data-wovn-ignore]';
  const TICKET_SELECTOR = '.my-ticket-info__name';
  const TOP_SELECTOR = '.my-top__heading01 span[data-wovn-ignore]';
  const MIRROR_SELECTORS = [HEADER_SELECTOR, TOP_SELECTOR];
  const EMPTY_DATA = { '氏名': '', '生年月日': '', '性別': '' };
  
  let pageType = null;
  
  // ==================== Helper Functions ====================
  
  const checkActive = () => localStorage.getItem(TOGGLE_KEY) === 'true';
  const setActive = (val) => localStorage.setItem(TOGGLE_KEY, val ? 'true' : 'false');
  
  const getFieldByLabel = (label) => {
    const titles = document.querySelectorAll('.form__title');
    for (const t of titles) {
      if (t.textContent.trim() === label) {
        const f = t.nextElementSibling;
        if (f && f.classList.contains('form__field')) return f;
      }
    }
    return null;
  };
  
  const getTextNodeFromField = (field) => {
    for (const n of field.childNodes) {
      if (n.nodeType === 3 && n.textContent.trim()) return n;
    }
    return null;
  };
  
  const parseName = (fullName) => {
    if (!fullName) return ['', ''];
    const idx = fullName.search(/\s/);
    if (idx === -1) return [fullName.trim(), ''];
    return [fullName.slice(0, idx).trim(), fullName.slice(idx + 1).trim()];
  };
  
  const getFirstNameInput = () => {
    const blocks = document.querySelectorAll('.form-name__block');
    for (const b of blocks) {
      const lb = b.querySelector('.form-name__label');
      if (lb && lb.textContent.trim() === '名') 
        return b.querySelector('.form-name__field[data-wovn-ignore]');
    }
    return null;
  };
  
  const getNameNode = (container, which) => {
    const children = [...container.childNodes];
    for (let i = 0; i < children.length; i++) {
      const n = children[i];
      if (n.nodeType === 1 && n.classList.contains('my-ticket-info__name-title') 
           && n.textContent.trim() === which) {
        const nx = children[i + 1];
        if (nx && nx.nodeType === 3) return nx;
      }
    }
    return null;
  };
  
  // ==================== Mirror Functions ====================
  
  const updateMirrors = (replacements) => {
    MIRROR_SELECTORS.forEach(sel => {
      document.querySelectorAll(sel).forEach(el => {
        if (el.hasAttribute('data-replaced')) return;
        if (!el.hasAttribute('data-original')) 
          el.setAttribute('data-original', el.textContent.trim());
        const val = replacements['氏名'];
        if (val !== undefined && val.trim() !== '') el.textContent = val;
        el.setAttribute('data-replaced', 'true');
      });
    });
  };
  
  const markMirrors = () => {
    MIRROR_SELECTORS.forEach(sel => {
      document.querySelectorAll(sel).forEach(el => {
        if (!el.hasAttribute('data-replaced')) 
          el.setAttribute('data-replaced', 'true');
      });
    });
  };
  
  const restoreMirrors = () => {
    MIRROR_SELECTORS.forEach(sel => {
      document.querySelectorAll(sel).forEach(el => {
        if (!el.hasAttribute('data-replaced')) return;
        if (el.hasAttribute('data-original')) 
          el.textContent = el.getAttribute('data-original');
        el.removeAttribute('data-replaced');
      });
    });
  };
  
  const readMirror = () => {
    for (const sel of MIRROR_SELECTORS) {
      const el = document.querySelector(sel);
      if (el) return el.hasAttribute('data-original') 
        ? el.getAttribute('data-original') 
        : el.textContent.trim();
    }
    return undefined;
  };
  
  // ==================== CSS ====================
  
  const generateCSS = () => {
    const hideSel = [...MIRROR_SELECTORS, TICKET_SELECTOR].join(',');
    const showSel = [...MIRROR_SELECTORS.map(x => x + '[data-replaced="true"]'), 
                     TICKET_SELECTOR + '[data-replaced="true"]'].join(',');
    return hideSel + ':not([data-replaced="true"]) { opacity: 0 !important; }' 
         + showSel + '{ opacity: 1 !important; transition: opacity 0.2s ease }'
         + '.account-table__data:not([data-replaced="true"]) { opacity: 0 !important; }'
         + '.account-table__data[data-replaced="true"] { opacity: 1 !important; transition: opacity 0.2s ease }';
  };
  
  const insertCSS = () => {
    const root = document.documentElement;
    if (root) {
      if (!root.querySelector('style[data-css-ok]')) {
        const style = document.createElement('style');
        style.textContent = generateCSS();
        style.setAttribute('data-css-ok', '1');
        root.insertBefore(style, root.firstChild);
      }
      return true;
    }
    return false;
  };
  
  const initCSS = () => {
    if (insertCSS()) return;
    let tries = 0;
    const retry = () => {
      if (insertCSS()) return;
      if (tries++ < 60) requestAnimationFrame(retry);
    };
    requestAnimationFrame(retry);
  };
  
  initCSS();
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initCSS);
  }
  
  // ==================== Page Visibility ====================
  
  const hidePage = () => {
    if (document.documentElement) 
      document.documentElement.style.setProperty('display', 'none', 'important');
  };
  
  const showPage = () => {
    if (document.documentElement) 
      document.documentElement.style.removeProperty('display');
  };
  
  hidePage();
  if (!document.documentElement) {
    let t = 0;
    const loop = () => {
      hidePage();
      if (document.documentElement) return;
      if (t++ < 60) requestAnimationFrame(loop);
    };
    requestAnimationFrame(loop);
  }
  
  // ==================== Data Storage ====================
  
  const readData = () => {
    try {
      const saved = localStorage.getItem(DATA_KEY);
      if (saved) return JSON.parse(saved);
    } catch(e) {}
    return { ...EMPTY_DATA };
  };
  
  const writeData = (data) => {
    try {
      localStorage.setItem(DATA_KEY, JSON.stringify(data));
    } catch(e) {}
  };
  
  const escapeHtml = (text) => {
    const div = document.createElement('div');
    div.textContent = text;
    return div.innerHTML;
  };
  
  // ==================== Form Guard ====================
  
  const guardForm = () => {
    if (pageType !== 'edit') return;
    const input = document.querySelector('input[name="user[last_name]"]');
    const form = input && input.closest('form');
    if (form && !form.hasAttribute('data-lp-guard')) {
      form.setAttribute('data-lp-guard', 'true');
      form.addEventListener('submit', () => {
        if (input.hasAttribute('data-original')) 
          input.value = input.getAttribute('data-original');
      }, true);
    }
  };
  
  // ==================== Extract Data ====================
  
  const pullData = () => {
    const result = {};
    
    if (pageType === 'view') {
      document.querySelectorAll('.account-table__block').forEach(block => {
        const h = block.querySelector('h3.account-table__title');
        const p = block.querySelector('p.account-table__data');
        if (!h || !p) return;
        const label = h.textContent.trim();
        if (!FIELD_LIST.includes(label)) return;
        result[label] = p.hasAttribute('data-original') 
          ? p.getAttribute('data-original') 
          : p.textContent.trim();
      });
    } else if (pageType === 'edit') {
      const input = document.querySelector('input[name="user[last_name]"]');
      if (input) result['氏名'] = input.hasAttribute('data-original') 
        ? input.getAttribute('data-original') 
        : input.value.trim();
      
      ['性別', '生年月日'].forEach(label => {
        const field = getFieldByLabel(label);
        if (!field) return;
        const tn = getTextNodeFromField(field);
        if (tn) result[label] = field.hasAttribute('data-original') 
          ? field.getAttribute('data-original') 
          : tn.textContent.trim();
      });
    }
    
    if (result['氏名'] === undefined) {
      const ticket = document.querySelector(TICKET_SELECTOR);
      if (ticket) {
        const tn = getNameNode(ticket, '氏');
        if (tn) result['氏名'] = ticket.hasAttribute('data-original') 
          ? ticket.getAttribute('data-original') 
          : tn.textContent.trim();
      }
    }
    
    if (result['氏名'] === undefined) {
      const mirror = readMirror();
      if (mirror) result['氏名'] = mirror;
    }
    
    return result;
  };
  
  // ==================== Mark Processed ====================
  
  const markDone = () => {
    if (pageType === 'view') {
      document.querySelectorAll('.account-table__data').forEach(p => {
        if (!p.hasAttribute('data-replaced')) 
          p.setAttribute('data-replaced', 'true');
      });
    } else if (pageType === 'edit') {
      const input = document.querySelector('input[name="user[last_name]"]');
      if (input && !input.hasAttribute('data-replaced')) 
        input.setAttribute('data-replaced', 'true');
      
      const firstName = getFirstNameInput();
      if (firstName && !firstName.hasAttribute('data-replaced')) 
        firstName.setAttribute('data-replaced', 'true');
      
      ['性別', '生年月日'].forEach(label => {
        const field = getFieldByLabel(label);
        if (!field || field.hasAttribute('data-replaced')) return;
        field.setAttribute('data-replaced', 'true');
      });
    }
    
    document.querySelectorAll(TICKET_SELECTOR).forEach(s => {
      if (!s.hasAttribute('data-replaced')) 
        s.setAttribute('data-replaced', 'true');
    });
    
    markMirrors();
  };
  
  // ==================== Apply ====================
  
  const applyAll = () => {
    const rep = readData();
    
    if (pageType === 'view') {
      document.querySelectorAll('.account-table__block').forEach(block => {
        const h = block.querySelector('h3.account-table__title');
        const p = block.querySelector('p.account-table__data');
        if (!h || !p) return;
        const label = h.textContent.trim();
        if (!FIELD_LIST.includes(label)) {
          if (!p.hasAttribute('data-replaced')) 
            p.setAttribute('data-replaced', 'true');
          return;
        }
        if (p.hasAttribute('data-replaced')) return;
        if (!p.hasAttribute('data-original')) 
          p.setAttribute('data-original', p.textContent.trim());
        
        const val = rep[label];
        if (val !== undefined && val.trim() !== '') p.textContent = val;
        p.setAttribute('data-replaced', 'true');
      });
      
      document.querySelectorAll('.account-table__data:not([data-replaced])')
        .forEach(p => p.setAttribute('data-replaced', 'true'));
        
    } else if (pageType === 'edit') {
      guardForm();
      
      const input = document.querySelector('input[name="user[last_name]"]');
      if (input && !input.hasAttribute('data-replaced')) {
        if (!input.hasAttribute('data-original')) 
          input.setAttribute('data-original', input.value.trim());
        
        const val = rep['氏名'];
        if (val !== undefined && val.trim() !== '') {
          const parts = parseName(val);
          input.value = parts[0];
          
          const firstName = getFirstNameInput();
          if (firstName) {
            if (!firstName.hasAttribute('data-original')) 
              firstName.setAttribute('data-original', firstName.textContent.trim());
            if (parts[1]) firstName.textContent = parts[1];
            firstName.setAttribute('data-replaced', 'true');
          }
        }
        input.setAttribute('data-replaced', 'true');
      }
      
      ['性別', '生年月日'].forEach(label => {
        const field = getFieldByLabel(label);
        if (!field || field.hasAttribute('data-replaced')) return;
        
        const tn = getTextNodeFromField(field);
        if (!tn) return;
        
        if (!field.hasAttribute('data-original')) 
          field.setAttribute('data-original', tn.textContent.trim());
        
        const val = rep[label];
        if (val !== undefined && val.trim() !== '') tn.textContent = val;
        field.setAttribute('data-replaced', 'true');
      });
    }
    
    document.querySelectorAll(TICKET_SELECTOR).forEach(s => {
      if (s.hasAttribute('data-replaced')) return;
      
      const shi = getNameNode(s, '氏');
      const mei = getNameNode(s, '名');
      if (!shi) return;
      
      if (!s.hasAttribute('data-original')) 
        s.setAttribute('data-original', 
          shi.textContent.trim() + '\u0000' + (mei ? mei.textContent.trim() : ''));
      
      const val = rep['氏名'];
      if (val !== undefined && val.trim() !== '') {
        const parts = parseName(val);
        shi.textContent = parts[0];
        if (mei && parts[1]) mei.textContent = parts[1];
      }
      s.setAttribute('data-replaced', 'true');
    });
    
    updateMirrors(rep);
  };
  
  // ==================== Reset ====================
  
  const resetAll = () => {
    if (pageType === 'view') {
      document.querySelectorAll('.account-table__data[data-replaced]').forEach(p => {
        if (p.hasAttribute('data-original')) 
          p.textContent = p.getAttribute('data-original');
        p.removeAttribute('data-replaced');
      });
    } else if (pageType === 'edit') {
      const input = document.querySelector('input[name="user[last_name]"]');
      if (input && input.hasAttribute('data-replaced')) {
        if (input.hasAttribute('data-original')) 
          input.value = input.getAttribute('data-original');
        input.removeAttribute('data-replaced');
      }
      
      const firstName = getFirstNameInput();
      if (firstName && firstName.hasAttribute('data-replaced')) {
        if (firstName.hasAttribute('data-original')) 
          firstName.textContent = firstName.getAttribute('data-original');
        firstName.removeAttribute('data-replaced');
      }
      
      ['性別', '生年月日'].forEach(label => {
        const field = getFieldByLabel(label);
        if (!field || !field.hasAttribute('data-replaced')) return;
        const tn = getTextNodeFromField(field);
        if (tn && field.hasAttribute('data-original')) 
          tn.textContent = field.getAttribute('data-original');
        field.removeAttribute('data-replaced');
      });
    }
    
    document.querySelectorAll(TICKET_SELECTOR).forEach(s => {
      if (!s.hasAttribute('data-replaced')) return;
      const shi = getNameNode(s, '氏');
      const mei = getNameNode(s, '名');
      const orig = s.getAttribute('data-original');
      if (orig !== null) {
        const parts = orig.split('\u0000');
        if (shi) shi.textContent = parts[0];
        if (mei && parts.length > 1) mei.textContent = parts[1];
      }
      s.removeAttribute('data-replaced');
    });
    
    restoreMirrors();
    applyAll();
  };
  
  // ==================== Settings Panel ====================
  
  const showPanel = () => {
    const existing = document.getElementById('lp-panel');
    if (existing) existing.remove();
    
    const data = readData();
    const on = checkActive();
    
    const panel = document.createElement('div');
    panel.id = 'lp-panel';
    
    const genderOptions = ['未选择', '男性', '女性'];
    
    panel.innerHTML = `
      <div style="position:fixed;inset:0;background:rgba(0,0,0,0.6);z-index:999998" id="lp-bg"></div>
      <div style="position:fixed;top:50%;left:50%;transform:translate(-50%,-50%);background:white;border-radius:16px;padding:24px;width:340px;max-height:85vh;overflow-y:auto;z-index:999999;box-shadow:0 20px 60px rgba(0,0,0,0.5);font-family:'Helvetica Neue',Arial,sans-serif">
        <div style="text-align:center;margin-bottom:16px">
          <h3 style="margin:0 0 8px;font-size:20px;color:#333">LivePocket 显示设置</h3>
          <div style="display:inline-block;padding:6px 12px;border-radius:20px;font-size:11px;font-weight:600;background:${on ? '#e8f5e9' : '#f5f5f5'};color:${on ? '#4caf50' : '#999'}">
            ${on ? '已启用' : '未启用'}
          </div>
        </div>
        
        ${FIELD_LIST.filter(f => f !== '性別').map(field => `
          <div style="margin-bottom:12px">
            <label style="display:block;font-size:12px;font-weight:600;color:#666;margin-bottom:6px">${field}</label>
            <input type="text" class="lp-input-field" data-label="${field}" value="${escapeHtml(data[field] || '')}" placeholder="留空则保持原样" style="width:100%;padding:10px;border:2px solid #e0e0e0;border-radius:8px;font-size:13px;box-sizing:border-box">
          </div>
        `).join('')}
        
        <div style="margin-bottom:16px">
          <label style="display:block;font-size:12px;font-weight:600;color:#666;margin-bottom:6px">性別</label>
          <div style="display:flex;gap:12px">
            ${genderOptions.map(opt => `
              <label style="display:flex;align-items:center;gap:4px;cursor:pointer;font-size:13px;color:#333">
                <input type="radio" name="lp-gender" value="${opt}" ${data['性別'] === opt || (!data['性別'] && opt === '未选择') ? 'checked' : ''}>
                <span>${opt}</span>
              </label>
            `).join('')}
          </div>
        </div>
        
        <div style="display:flex;gap:8px;justify-content:center;margin-top:16px;padding-top:16px;border-top:1px solid #eee">
          <button id="lp-btn-save" style="flex:1;padding:10px;background:#4caf50;color:white;border:none;border-radius:8px;cursor:pointer;font-weight:600;font-size:12px">保存</button>
          <button id="lp-btn-close" style="flex:1;padding:10px;background:#9e9e9e;color:white;border:none;border-radius:8px;cursor:pointer;font-weight:600;font-size:12px">取消</button>
        </div>
      </div>
    `;
    
    document.body.appendChild(panel);
    
    // Events
    document.getElementById('lp-bg').addEventListener('click', () => panel.remove());
    document.getElementById('lp-btn-close').addEventListener('click', () => panel.remove());
    
    document.getElementById('lp-btn-save').addEventListener('click', () => {
      const newData = {};
      panel.querySelectorAll('.lp-input-field').forEach(input => {
        newData[input.dataset.label] = input.value.trim();
      });
      const genderRadio = panel.querySelector('input[name="lp-gender"]:checked');
      if (genderRadio) {
        newData['性別'] = genderRadio.value;
      }
      writeData(newData);
      setActive(true);
      resetAll();
      panel.remove();
    });
  };
  
  // ==================== Double-click ====================
  
  const initDblClick = () => {
    const h1 = document.querySelector('h1.heading01');
    if (h1 && !h1.hasAttribute('data-lp-dbl')) {
      h1.style.cursor = 'pointer';
      h1.title = '双击打开设置';
      h1.addEventListener('dblclick', (e) => {
        e.preventDefault();
        showPanel();
      });
      h1.setAttribute('data-lp-dbl', '1');
    }
  };
  
  // ==================== Main Logic ====================
  
  const process = () => {
    pageType = document.querySelector('.account-table__block') 
      ? 'view' 
      : (document.querySelector('input[name="user[last_name]"]') ? 'edit' : null);
    
    if (checkActive()) applyAll();
    else markDone();
  };
  
  let obsStarted = false;
  const startObs = () => {
    if (obsStarted) return;
    obsStarted = true;
    
    let rid = null;
    new MutationObserver(ms => {
      if (!ms.some(m => m.addedNodes.length > 0)) return;
      if (rid) cancelAnimationFrame(rid);
      rid = requestAnimationFrame(() => {
        process();
        initDblClick();
        rid = null;
      });
    }).observe(document.documentElement || document, {
      childList: true,
      subtree: true
    });
  };
  
  // ==================== Init ====================
  
  let inited = false;
  let revealed = false;
  let tries = 0;
  
  const doReveal = () => {
    if (revealed) return;
    revealed = true;
    showPage();
  };
  
  setTimeout(doReveal, 1500);
  
  const run = () => {
    const p = document.querySelector('.account-table__block') 
      ? 'view' 
      : (document.querySelector('input[name="user[last_name]"]') ? 'edit' : null);
    
    const hasExtra = !!document.querySelector(HEADER_SELECTOR) 
                  || !!document.querySelectorAll(TICKET_SELECTOR).length;
    
    if (!p && !hasExtra) {
      if (!inited) {
        tries++;
        if (tries > 90) { doReveal(); return; }
        requestAnimationFrame(run);
      }
      return;
    }
    
    pageType = p;
    guardForm();
    
    if (!inited) {
      inited = true;
      
      if (!localStorage.getItem(DATA_KEY)) {
        const pd = pullData();
        writeData(Object.keys(pd).length > 0 ? pd : { ...EMPTY_DATA });
        markDone();
      } else if (checkActive()) {
        applyAll();
      } else {
        markDone();
      }
      
      initDblClick();
      startObs();
      doReveal();
    } else {
      if (checkActive()) {
        applyAll();
      }
    }
  };
  
  run();
  
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', run);
  }
  
  const loop = () => {
    if (!inited) {
      run();
      requestAnimationFrame(loop);
    }
  };
  
  if (!inited) {
    requestAnimationFrame(loop);
  }
  
})();
