// ==UserScript==
// @name         万代姓名修改
// @namespace    Andy LI
// @version      4.5
// @description  lanlan
// @match        *://*/*
// @grant        none
// @run-at       document-end
// ==/UserScript==

(function() {
    'use strict';

    let myCustomName = localStorage.getItem('STAY_CUSTOM_VISUAL_NAME') || '';
    let myCustomKana = localStorage.getItem('STAY_CUSTOM_VISUAL_KANA') || '';
    let myCustomBirth = localStorage.getItem('STAY_CUSTOM_VISUAL_BIRTH') || '';
    let myCustomGender = localStorage.getItem('STAY_CUSTOM_VISUAL_GENDER') || '';

    // 随机示例
    const randomSamples = {
        firstNames: ['AKIRA', 'YUKI', 'HARU', 'NANA', 'SORA', 'MIO', 'KEI', 'REI', 'TAI', 'YUMA'],
        lastNames: ['TANAKA', 'YAMADA', 'SUZUKI', 'WATANABE', 'ITO', 'NAKAMURA', 'KOBAYASHI', 'KATO'],
        kanaFirst: ['アキラ', 'ユキ', 'ハル', 'ナナ', 'ソラ', 'ミオ', 'ケイ', 'レイ', 'タイ', 'ユマ'],
        kanaLast: ['タナカ', 'ヤマダ', 'スズキ', 'ワタナベ', 'イトウ', 'ナカムラ', 'コバヤシ', 'カトウ']
    };

    function getRandomSample() {
        const first = randomSamples.firstNames[Math.floor(Math.random() * randomSamples.firstNames.length)];
        const last = randomSamples.lastNames[Math.floor(Math.random() * randomSamples.lastNames.length)];
        const kanaFirst = randomSamples.kanaFirst[Math.floor(Math.random() * randomSamples.kanaFirst.length)];
        const kanaLast = randomSamples.kanaLast[Math.floor(Math.random() * randomSamples.kanaLast.length)];
        const year = 1980 + Math.floor(Math.random() * 25);
        const month = String(1 + Math.floor(Math.random() * 12)).padStart(2, '0');
        const day = String(1 + Math.floor(Math.random() * 28)).padStart(2, '0');
        return {
            name: first + ' ' + last,
            kana: kanaFirst + ' ' + kanaLast,
            birth: year + '/' + month + '/' + day
        };
    }

    function convertRomajiToKatakana(romajiText) {
        if (!romajiText) return "";
        let text = romajiText.toUpperCase().trim();

        const mapping = {
            'A': 'ア', 'I': 'イ', 'U': 'ウ', 'E': 'エ', 'O': 'オ',
            'KA': 'カ', 'KI': 'キ', 'KU': 'ク', 'KE': 'ケ', 'KO': 'コ',
            'SA': 'サ', 'SHI': 'シ', 'SU': 'ス', 'SE': 'セ', 'SO': 'ソ',
            'TA': 'タ', 'CHI': 'チ', 'TSU': 'ツ', 'TE': 'テ', 'TO': 'ト',
            'NA': 'ナ', 'NI': 'ニ', 'NU': 'ヌ', 'NE': 'ネ', 'NO': 'ノ',
            'HA': 'ハ', 'HI': 'ヒ', 'FU': 'フ', 'HE': 'ヘ', 'HO': 'ホ',
            'MA': 'マ', 'MI': 'ミ', 'MU': 'ム', 'ME': 'メ', 'MO': 'モ',
            'YA': 'ヤ', 'YU': 'ユ', 'YO': 'ヨ',
            'RA': 'ラ', 'RI': 'リ', 'RU': 'ル', 'RE': 'レ', 'RO': 'ロ',
            'WA': 'ワ'
        };

        let words = text.split(/[\s　]+/);
        let resultWords = words.map(word => {
            let kana = ""; let i = 0;
            while (i < word.length) {
                let matched = false;
                for (let len = 4; len > 0; len--) {
                    if (i + len <= word.length) {
                        let substr = word.substr(i, len);
                        if (mapping[substr]) { kana += mapping[substr]; i += len; matched = true; break; }
                    }
                }
                if (!matched) i++;
            }
            return kana;
        });
        return resultWords.filter(w => w.length > 0).join(' ');
    }

    function getKatakanaName() {
        if (myCustomKana) return myCustomKana;
        return convertRomajiToKatakana(myCustomName);
    }

    // === 创建输入面板 ===
    function createInputPanel() {
        const sample = getRandomSample();

        // 移除已存在的面板
        const existing = document.getElementById('wandai-input-panel');
        if (existing) existing.remove();

        const panel = document.createElement('div');
        panel.id = 'wandai-input-panel';
        panel.innerHTML = `
            <div style="position:fixed;inset:0;background:rgba(0,0,0,0.5);z-index:999998" id="wandai-overlay"></div>
            <div style="position:fixed;top:50%;left:50%;transform:translate(-50%,-50%);background:white;border-radius:12px;padding:24px;width:360px;z-index:999999;box-shadow:0 10px 40px rgba(0,0,0,0.3);font-family:sans-serif">
                <h3 style="margin:0 0 20px;text-align:center;font-size:16px;color:#333">修改个人信息</h3>
                
                <div style="margin-bottom:14px">
                    <label style="display:block;font-size:12px;color:#666;margin-bottom:6px">英文名 <span style="color:#999;font-size:10px">示例: ${sample.name}</span></label>
                    <input type="text" id="wandai-name" value="${escapeHtml(myCustomName)}" placeholder="" style="width:100%;padding:10px;border:1px solid #ddd;border-radius:6px;font-size:13px;box-sizing:border-box">
                </div>
                
                <div style="margin-bottom:14px">
                    <label style="display:block;font-size:12px;color:#666;margin-bottom:6px">假名 <span style="color:#999;font-size:10px">示例: ${sample.kana}</span></label>
                    <input type="text" id="wandai-kana" value="${escapeHtml(myCustomKana)}" placeholder="" style="width:100%;padding:10px;border:1px solid #ddd;border-radius:6px;font-size:13px;box-sizing:border-box">
                </div>
                
                <div style="margin-bottom:14px">
                    <label style="display:block;font-size:12px;color:#666;margin-bottom:6px">生年月日 <span style="color:#999;font-size:10px">示例: ${sample.birth}</span></label>
                    <input type="text" id="wandai-birth" value="${escapeHtml(myCustomBirth)}" placeholder="" style="width:100%;padding:10px;border:1px solid #ddd;border-radius:6px;font-size:13px;box-sizing:border-box">
                </div>
                
                <div style="margin-bottom:20px">
                    <label style="display:block;font-size:12px;color:#666;margin-bottom:6px">性别</label>
                    <select id="wandai-gender" style="width:100%;padding:10px;border:1px solid #ddd;border-radius:6px;font-size:13px;box-sizing:border-box;background:white">
                        <option value="">不修改</option>
                        <option value="男性">男性</option>
                        <option value="女性">女性</option>
                    </select>
                </div>
                
                <div style="display:flex;gap:10px;justify-content:center">
                    <button id="wandai-save" style="padding:10px 24px;background:#4caf50;color:white;border:none;border-radius:6px;cursor:pointer;font-size:13px">保存</button>
                    <button id="wandai-cancel" style="padding:10px 24px;background:#999;color:white;border:none;border-radius:6px;cursor:pointer;font-size:13px">取消</button>
                </div>
            </div>
        `;

        document.body.appendChild(panel);

        // 设置性别下拉框的默认选中值
        if (myCustomGender) {
            const genderSelect = document.getElementById('wandai-gender');
            for (let i = 0; i < genderSelect.options.length; i++) {
                if (genderSelect.options[i].value === myCustomGender) {
                    genderSelect.options[i].selected = true;
                    break;
                }
            }
        }

        // 事件绑定
        document.getElementById('wandai-overlay').addEventListener('click', () => panel.remove());
        document.getElementById('wandai-cancel').addEventListener('click', () => panel.remove());
        document.getElementById('wandai-save').addEventListener('click', () => {
            const name = document.getElementById('wandai-name').value.trim();
            const kana = document.getElementById('wandai-kana').value.trim();
            const birth = document.getElementById('wandai-birth').value.trim();
            const gender = document.getElementById('wandai-gender').value.trim();

            if (name) {
                myCustomName = name;
                localStorage.setItem('STAY_CUSTOM_VISUAL_NAME', myCustomName);
                // 自动转换假名（如果假名为空）
                if (!kana && !myCustomKana) {
                    myCustomKana = convertRomajiToKatakana(myCustomName);
                    localStorage.setItem('STAY_CUSTOM_VISUAL_KANA', myCustomKana);
                }
            }

            if (kana) {
                myCustomKana = kana;
                localStorage.setItem('STAY_CUSTOM_VISUAL_KANA', myCustomKana);
            } else if (!name) {
                // 如果没填英文名且清空了假名
                myCustomKana = '';
                localStorage.removeItem('STAY_CUSTOM_VISUAL_KANA');
            }

            if (birth) {
                myCustomBirth = birth;
                localStorage.setItem('STAY_CUSTOM_VISUAL_BIRTH', myCustomBirth);
            } else {
                myCustomBirth = '';
                localStorage.removeItem('STAY_CUSTOM_VISUAL_BIRTH');
            }

            if (gender) {
                myCustomGender = gender;
                localStorage.setItem('STAY_CUSTOM_VISUAL_GENDER', myCustomGender);
            } else {
                myCustomGender = '';
                localStorage.removeItem('STAY_CUSTOM_VISUAL_GENDER');
            }

            cleanWebPageVisuals();
            panel.remove();
        });
    }

    function escapeHtml(text) {
        const div = document.createElement('div');
        div.textContent = text;
        return div.innerHTML;
    }

    // === 核心清洗函数 ===
    function cleanWebPageVisuals() {
        if (!myCustomName && !myCustomBirth && !myCustomGender) return;

        const katakanaName = getKatakanaName();

        // 适配
        const couponNameNodes = document.querySelectorAll('.block-mypage-coupon-list-item-code-value');
        couponNameNodes.forEach(node => {
            if (myCustomName && node.textContent.trim() !== myCustomName && !node.textContent.includes('EC-')) {
                node.textContent = myCustomName;
            }
        });

        // 历史订单
        const contents = document.querySelectorAll('.block-mypage-history-block-detail-contents-block-content');
        contents.forEach(content => {
            let html = content.innerHTML;
            
            if (html.includes('<br') || html.includes('<BR')) {
                const lines = html.split(/<br>|<BR>/i);
                if (lines && lines.length > 0) {
                    const firstLineText = lines[0].trim();
                    const expectedLine = myCustomName + "(" + katakanaName + ")";
                    if (myCustomName && firstLineText !== myCustomName && firstLineText !== expectedLine && !html.includes('決済') && !html.includes('结算')) {
                        lines[0] = expectedLine; 
                        content.innerHTML = lines.join('<br>');
                    }
                }
            } else {
                const textTrim = content.textContent.trim();
                if (myCustomName && textTrim && textTrim !== myCustomName && !textTrim.includes('決済') && !textTrim.includes('结算') && textTrim.length < 30) {
                    content.innerHTML = myCustomName;
                }
            }
        });

        // 会员页面
        const labels = document.querySelectorAll('.block-mypage-member-info-label');
        labels.forEach(label => {
            const labelText = label.textContent.trim();
            if (labelText === '氏名（漢字）' || labelText === '氏名（カナ）' || labelText === '生年月日' || labelText === '性別') {
                const dlItem = label.closest('.block-mypage-member-info-item') || label.parentElement;
                const valueNode = dlItem ? dlItem.querySelector('.block-mypage-member-info-value') : null;

                if (valueNode) {
                    if (labelText === '氏名（漢字）' && myCustomName && valueNode.textContent.trim() !== myCustomName) {
                        valueNode.textContent = myCustomName;
                    }
                    if (labelText === '氏名（カナ）' && katakanaName && valueNode.textContent.trim() !== katakanaName) {
                        valueNode.textContent = katakanaName;
                    }
                    if (labelText === '生年月日' && myCustomBirth && valueNode.textContent.trim() !== myCustomBirth) {
                        valueNode.textContent = myCustomBirth;
                    }
                    if (labelText === '性別' && myCustomGender && valueNode.textContent.trim() !== myCustomGender) {
                        valueNode.textContent = myCustomGender;
                    }
                }
            }
        });
    }

    // 防抖节流
    let debounceTimer = null;
    function debounceWash() {
        clearTimeout(debounceTimer);
        debounceTimer = setTimeout(cleanWebPageVisuals, 20);
    }

    // 双击触发交互
    document.addEventListener('dblclick', function(event) {
        const text = event.target.textContent;

        if (text.includes('会員情報') || text.includes('会员') || text.includes('チケット') || text.includes('券')) {
            event.preventDefault();
            createInputPanel();
        }
    });

    // 监控驱动
    window.addEventListener('DOMContentLoaded', cleanWebPageVisuals);
    window.addEventListener('load', cleanWebPageVisuals);

    const observer = new MutationObserver(debounceWash);
    observer.observe(document.documentElement, { 
        childList: true, 
        subtree: true, 
        characterData: true 
    });

    cleanWebPageVisuals();
})();
