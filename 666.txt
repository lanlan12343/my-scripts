(function () {
  const NK = 'parks2_member_name';
  const KK = 'parks2_member_kana';
  const url = location.href;

  if (url.includes('member_mypage.html')) {
    const saveKana = () => {
      for (const dt of document.querySelectorAll('dt.block-mypage-member-info-label')) {
        if (dt.textContent.trim() === '氏名（カナ）') {
          const dd = dt.nextElementSibling;
          if (dd) {
            localStorage.setItem(KK, dd.textContent.trim());
            return true;
          }
        }
      }
      return false;
    };

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', saveKana);
    } else {
      saveKana();
    }

    let done = false;
    new MutationObserver((_, obs) => {
      if (!done && saveKana()) {
        done = true;
        obs.disconnect();
      }
    }).observe(document.documentElement, { childList: true, subtree: true });
  }

  if (url.includes('member_history.html')) {
    const name = localStorage.getItem(NK);
    const kana = localStorage.getItem(KK);
    if (!name && !kana) return;

    const replaceHistoryName = () => {
      document.querySelectorAll('dl.block-mypage-history-block-detail-contents-block').forEach((dl) => {
        const dt = dl.querySelector('dt.block-mypage-history-block-detail-contents-block-label');
        const dd = dl.querySelector('dd.block-mypage-history-block-detail-contents-block-content');
        if (!dt || !dd || dd.hasAttribute('data-history-replaced')) return;

        const label = dt.textContent.trim();

        if (label === 'ご依頼主') {
          if (!name) return;
          dd.setAttribute('data-original-html', dd.innerHTML);
          dd.textContent = name;
          dd.setAttribute('data-history-replaced', 'true');
          return;
        }

        if (label === 'お届け先') {
          const firstBr = Array.from(dd.childNodes).find((node) => node.nodeName === 'BR');
          const fullName = name ? (kana ? `${name}(${kana})` : name) : kana;

          if (!fullName) return;

          dd.setAttribute('data-original-html', dd.innerHTML);

          if (!firstBr) {
            dd.textContent = fullName;
          } else {
            while (dd.firstChild && dd.firstChild !== firstBr) {
              dd.removeChild(dd.firstChild);
            }
            dd.insertBefore(document.createTextNode(fullName), firstBr);
          }

          dd.setAttribute('data-history-replaced', 'true');
        }
      });
    };

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', replaceHistoryName);
    } else {
      replaceHistoryName();
    }

    new MutationObserver(replaceHistoryName).observe(document.documentElement, {
      childList: true,
      subtree: true
    });
  }
